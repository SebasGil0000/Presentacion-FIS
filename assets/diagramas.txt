@startuml
actor Usuario
participant "Interfaz (Frontend)" as Frontend
participant "Lógica de Comparación" as Logica
participant "Azure Speech API" as API

Usuario -> Frontend : Iniciar sesión de lectura
activate Frontend

Frontend -> Usuario : Solicitar acceso a micrófono
Usuario --> Frontend : Acceso concedido

Frontend -> Logica : Preparar entorno (ID Texto Base)
activate Logica
Logica --> Frontend : Entorno listo
deactivate Logica

alt Bucle de evaluación en tiempo real
    Usuario -> Frontend : Locución de texto (Audio)
    
    Frontend -> API : Transmisión de stream de audio
    activate API
    API --> Frontend : Retorno de transcripción parcial
    deactivate API

    Frontend -> Logica : Solicitud de validación de texto
    activate Logica
    Logica --> Frontend : Estado de palabra (Acierto/Error)
    deactivate Logica

    Frontend -> Usuario : Renderizar cambio de color en pantalla
end

Usuario -> Frontend : Finalizar lectura

Frontend -> Logica : Cerrar sesión de captura
activate Logica
Logica -> Logica : Procesar métricas finales
Logica --> Frontend : Objeto de resultados
deactivate Logica

Frontend -> Usuario : Visualizar reporte de desempeño
deactivate Frontend
@enduml

class Usuario {
  + id : UUID
  + correoElectronico : String
  - nombreCompleto : String
  + iniciarSesion()
}

class SesionEvaluacion {
  + idSesion : UUID
  + fechaInicio : DateTime
  + totalAciertos : int
  + totalErrores : int
  + calcularPorcentajePrecision() : float
  + generarReporteDesempeno() : Reporte
}

class TextoEstudio {
  + idTexto : int
  + titulo : String
  + contenidoBase : String
  + palabras : List<String>
  + obtenerPalabraActual(indice: int)
}

class AlgoritmoComparacion {
  - textoReferencia : TextoEstudio
  + evaluarCoincidenciaExacta(palabraHablada: String, palabraEsperada: String) : boolean
  + sincronizarFlujo(transcripcion: String)
}

interface MotorReconocimientoVoz {
  + iniciarStreamAudio()
  + detenerStreamAudio()
}

class ServicioAzureSpeech {
  - claveAcceso : String
  - regionDespliegue : String
  + establecerConexion()
  - convertirAudioATexto(flujo: AudioStream) : String
}

Usuario "1" -- "*" SesionEvaluacion : ejecuta >
SesionEvaluacion "*" -- "1" TextoEstudio : requiere >
SesionEvaluacion "1" -- "1" AlgoritmoComparacion : delega analisis a >
AlgoritmoComparacion ..> MotorReconocimientoVoz : consume >
ServicioAzureSpeech ..|> MotorReconocimientoVoz : implementa
@enduml
